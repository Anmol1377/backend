MODULE 01

Speak HTTP

HTTP is just structured text. A client writes a method (what to do), a path (to what), and headers (context) — the server writes back a status line, headers and a body. Everything else in backend engineering is layered on this conversation.

The four verbs you'll use daily: GET reads, POST creates, PUT replaces, DELETE removes. GET should never change data — that rule is what makes caching and retries safe.

Try: POST to /login with the wrong body, then DELETE a user that doesn't exist. Read the raw response each time.

// npm i express — the most common Node web framework
import express from "express";
const app = express();
app.use(express.json());              // parse JSON request bodies

app.get("/users/:id", (req, res) => {
  const user = db.find(req.params.id);
  if (!user) return res.status(404).json({ error: "not found" });
  res.json(user);                     // 200 OK + JSON body
});

app.post("/users", (req, res) => {
  const created = db.insert(req.body);
  res.status(201)                     // 201 Created
     .location(`/users/${created.id}`)// tell the client where it lives
     .json(created);
});

app.delete("/users/:id", (req, res) => {
  db.remove(req.params.id);
  res.status(204).end();              // 204 No Content — empty body
});

app.listen(3000, () => console.log("API on http://localhost:3000"));
// Go 1.22+ standard library — no framework needed
package main

import (
	"encoding/json"
	"net/http"
)

func main() {
	mux := http.NewServeMux()

	// method + path patterns are built in since Go 1.22
	mux.HandleFunc("GET /users/{id}", func(w http.ResponseWriter, r *http.Request) {
		user, ok := db.Find(r.PathValue("id"))
		if !ok {
			writeJSON(w, http.StatusNotFound, map[string]string{"error": "not found"})
			return
		}
		writeJSON(w, http.StatusOK, user) // 200
	})

	mux.HandleFunc("POST /users", func(w http.ResponseWriter, r *http.Request) {
		var in User
		json.NewDecoder(r.Body).Decode(&in)
		created := db.Insert(in)
		w.Header().Set("Location", "/users/"+created.ID)
		writeJSON(w, http.StatusCreated, created) // 201
	})

	http.ListenAndServe(":3000", mux)
}

func writeJSON(w http.ResponseWriter, code int, v any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(code)
	json.NewEncoder(w).Encode(v)
}

How to run it

Node: save as server.mjs, then node server.mjs. Test with curl -X POST localhost:3000/users -d '{"name":"Ada"}' -H "content-type: application/json".
Go: save as main.go, then go run main.go. Same curl.

Going further

Validate input before trusting it — zod (Node) or go-playground/validator (Go). Make writes idempotent so a client retry can't create two users. Never build responses by string-concatenating user input (see the SQL injection module). In production you'd add middleware for logging, auth (JWT module) and rate limiting.

🔧 Go deeper — set status codes correctly in code

Returning the right code is a one-liner — the skill is choosing it. These handlers show the common ones in context.

app.post("/orders", (req, res) => {
  if (!req.body.item)                       // client forgot a field
    return res.status(400).json({ error: "item is required" });

  if (!req.user)                            // no valid token
    return res.status(401).json({ error: "log in first" });

  if (req.user.role !== "buyer")            // logged in, not allowed
    return res.status(403).json({ error: "buyers only" });

  if (db.soldOut(req.body.item))            // conflicts with state
    return res.status(409).json({ error: "sold out" });

  const order = db.createOrder(req.user, req.body.item);
  res.status(201).json(order);              // success — created
});

// global safety net: any thrown error becomes a 500
app.use((err, req, res, next) => {
  console.error(err);                       // read THIS when debugging
  res.status(500).json({ error: "something broke on our end" });
});
func createOrder(w http.ResponseWriter, r *http.Request) {
	var in OrderInput
	if err := json.NewDecoder(r.Body).Decode(&in); err != nil || in.Item == "" {
		http.Error(w, "item is required", http.StatusBadRequest) // 400
		return
	}
	user, ok := userFromContext(r) // set by your auth middleware
	if !ok {
		http.Error(w, "log in first", http.StatusUnauthorized) // 401
		return
	}
	if user.Role != "buyer" {
		http.Error(w, "buyers only", http.StatusForbidden) // 403
		return
	}
	if db.SoldOut(in.Item) {
		http.Error(w, "sold out", http.StatusConflict) // 409
		return
	}
	order := db.CreateOrder(user, in.Item)
	writeJSON(w, http.StatusCreated, order) // 201
}

Going further

Prefer named constants (http.StatusNotFound, not 404) — they read better and never typo. Put one error-handling middleware at the end so an unexpected throw becomes a clean 500 with a logged stack trace, never a leaked one. For redirects, set the Location header and the 301/302 code together.
MODULE 03

Databases & the index race

A query without an index reads every row until it finds a match — O(n). An index is a sorted tree kept next to the table: the database hops down it in O(log n) steps instead.

At 100 rows nobody notices. At a million rows it's the difference between microseconds and seconds — and between a calm on-call week and a pager going off.

Try: race at 100 rows, then drag to 5,000. The scan's cost grows linearly; the index barely moves. That gap is the whole story of database performance.

🔧 Go deeper — query with a real database & add the index

The race showed why the index matters. Here's the actual code — a parameterized query against PostgreSQL, plus the one-line migration that makes it fast.

// npm i pg
import { Pool } from "pg";
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

// $1 is a placeholder — the driver sends value separately (safe!)
async function findUserByEmail(email) {
  const { rows } = await pool.query(
    "SELECT id, name FROM users WHERE email = $1",
    [email]
  );
  return rows[0];
}

// Without an index this SELECT scans every row: O(n).
// Run this once (as a migration) and it becomes O(log n):
//   CREATE INDEX CONCURRENTLY idx_users_email ON users(email);

// See the query plan yourself — look for "Index Scan" not "Seq Scan":
//   EXPLAIN ANALYZE SELECT ... WHERE email = 'a@b.com';
// go get github.com/jackc/pgx/v5/stdlib
import (
	"context"
	"database/sql"
)

func findUserByEmail(ctx context.Context, db *sql.DB, email string) (User, error) {
	var u User
	// $1 placeholder — pgx never interpolates the value into the SQL text
	err := db.QueryRowContext(ctx,
		"SELECT id, name FROM users WHERE email = $1", email,
	).Scan(&u.ID, &u.Name)
	return u, err
}

// The migration that turns the Seq Scan into an Index Scan:
//   CREATE INDEX CONCURRENTLY idx_users_email ON users(email);
//
// Verify it's used:
//   EXPLAIN ANALYZE SELECT id, name FROM users WHERE email = 'a@b.com';

Going further

Index the columns you filter and join on — but not every column: each index slows down writes and costs disk. Composite indexes like (tenant_id, created_at) serve queries that filter on both. Always use a connection pool (shown above) — opening a fresh connection per request is a classic performance bug. Reach for an ORM (Prisma, TypeORM, GORM, sqlc) once raw SQL gets repetitive, but keep reading the SQL it generates.
MODULE 04

Caching & LRU eviction

A cache keeps hot data in fast memory so you skip the database. But memory is small, so when it's full something must go. LRU — evict the least recently used item — is the classic policy, because "recently touched" is the best cheap prediction of "about to be touched again".

This cache holds 4 keys. Every hit moves a key to the front; every miss goes to the (slow) database and pushes out whatever sits at the back. In real backends this cache is usually Redis — an in-memory key-value store (a giant hash table, see FIG. 11) shared by all your servers.

Try: hammer two users repeatedly — hit rate soars. Then request all six in a loop and watch the cache thrash: everything evicts everything.

🔧 Go deeper — the cache-aside pattern with Redis

The LRU figure lives inside Redis in the real world. This is the cache-aside pattern: check cache → miss → hit the DB → store with a TTL. It's the single most common caching code you'll write.

// npm i ioredis
import Redis from "ioredis";
const redis = new Redis(process.env.REDIS_URL);

async function getUser(id) {
  const key = `user:${id}`;

  const cached = await redis.get(key);        // 1. try cache
  if (cached) return JSON.parse(cached);      //    HIT — ~0.5ms, done

  const user = await db.findUser(id);         // 2. MISS — hit the DB (~10ms)
  await redis.set(key, JSON.stringify(user),
                  "EX", 60);                  // 3. cache for 60s (TTL)
  return user;
}

// When the user changes, delete the key so the next read repopulates:
async function updateUser(id, data) {
  await db.updateUser(id, data);
  await redis.del(`user:${id}`);              // invalidate — avoid stale reads
}
// go get github.com/redis/go-redis/v9
import (
	"context"
	"encoding/json"
	"time"
)

func getUser(ctx context.Context, id string) (User, error) {
	key := "user:" + id

	if b, err := rdb.Get(ctx, key).Bytes(); err == nil { // 1. try cache
		var u User
		json.Unmarshal(b, &u)
		return u, nil                                     //    HIT
	}

	u, err := db.FindUser(ctx, id)                        // 2. MISS -> DB
	if err != nil {
		return User{}, err
	}
	b, _ := json.Marshal(u)
	rdb.Set(ctx, key, b, 60*time.Second)                  // 3. cache 60s TTL
	return u, nil
}

func updateUser(ctx context.Context, id string, d Data) error {
	if err := db.UpdateUser(ctx, id, d); err != nil {
		return err
	}
	return rdb.Del(ctx, "user:"+id).Err()                 // invalidate
}

Going further

Always set a TTL — a cache with no expiry is a memory leak and a source of stale data. Invalidate on write (delete the key) so readers don't see old values. Beware the thundering herd: if a hot key expires under load, thousands of requests stampede the DB at once — mitigate with a short lock or "stale-while-revalidate". Redis is also where your rate-limiter buckets and session data live.
MODULE 05

Load balancing

One server has a ceiling. A load balancer puts N servers behind one address and spreads requests across them. Round robin deals them like cards; least connections sends each request to whoever is idlest.

The quiet superpower is health checks: when a server stops answering, the balancer simply stops sending traffic to it. Users never know.

Try: start traffic, then kill a server. Watch its load drain to zero while the others absorb the flow — that's failover, the reason deploys don't cause outages.

🔧 Go deeper — a health check + reverse-proxy load balancer

Usually nginx or your cloud does this. But it's ~40 lines, and writing it once demystifies the whole thing: a pool of backends, a health check that ejects the dead one, and round-robin routing.

import http from "http";
import httpProxy from "http-proxy"; // npm i http-proxy

const backends = [
  { url: "http://localhost:4001", alive: true },
  { url: "http://localhost:4002", alive: true },
  { url: "http://localhost:4003", alive: true },
];
const proxy = httpProxy.createProxyServer();
let i = 0;

// health check: every 5s, mark unresponsive backends dead
setInterval(async () => {
  for (const b of backends) {
    try { await fetch(b.url + "/health"); b.alive = true; }
    catch { b.alive = false; }               // stop routing to it
  }
}, 5000);

http.createServer((req, res) => {
  const healthy = backends.filter(b => b.alive);
  if (!healthy.length) { res.writeHead(503).end("no backends"); return; }
  const target = healthy[i++ % healthy.length].url; // round robin
  proxy.web(req, res, { target }, () => res.writeHead(502).end());
}).listen(8080, () => console.log("LB on :8080"));
package main

import (
	"net/http"
	"net/http/httputil"
	"net/url"
	"sync/atomic"
	"time"
)

var backends = []string{"http://localhost:4001", "http://localhost:4002"}
var counter uint64

func main() {
	// health checks would flip a per-backend "alive" flag on a ticker;
	// omitted here for space — same idea as the Node version.
	http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
		n := atomic.AddUint64(&counter, 1)
		target, _ := url.Parse(backends[n%uint64(len(backends))]) // round robin
		httputil.NewSingleHostReverseProxy(target).ServeHTTP(w, r)
	})
	srv := &http.Server{Addr: ":8080", ReadTimeout: 5 * time.Second}
	srv.ListenAndServe()
}

Going further

In practice you rarely write this — you configure nginx, HAProxy, or a cloud load balancer (AWS ALB, GCP LB). The lesson that transfers: your app servers must be stateless (no in-memory sessions — use JWTs or Redis) so any request can go to any backend. That's what makes both load balancing and zero-downtime deploys possible.
MODULE 06

Message queues

Some work shouldn't happen during the request — sending emails, resizing images, generating reports. A queue lets the API say "accepted!" instantly (202), drop a job in the buffer, and let workers chew through it at their own pace.

The queue's depth is the most honest metric in your system: producers faster than consumers ⇒ depth grows ⇒ you need more workers, or you're about to have a bad day.

Try: push the producer to 8 jobs/s with one worker and watch the backlog explode. Then add workers until depth drains. You just did capacity planning.

🔧 Go deeper — producer & worker with a real queue

The figure showed depth growing and workers draining it. Here's the code on both sides: an API that enqueues a job and returns 202 instantly, and a worker process that consumes.

// npm i bullmq — a Redis-backed job queue
import { Queue, Worker } from "bullmq";
const connection = { host: "localhost", port: 6379 };

// --- PRODUCER: inside your API request handler ---
const emailQueue = new Queue("emails", { connection });

app.post("/signup", async (req, res) => {
  const user = await db.createUser(req.body); // the fast, critical part
  await emailQueue.add("welcome", { to: user.email }); // drop a job
  res.status(202).json({ status: "accepted" }); // reply NOW, ~15ms
});

// --- CONSUMER: a separate worker process ---
new Worker("emails", async (job) => {
  await sendEmail(job.data.to);   // the slow part, off the request path
}, { connection, concurrency: 5 }); // 5 jobs in parallel

// Scaling = run more worker processes. BullMQ retries failed jobs.
// For production: github.com/hibiken/asynq (Redis-backed).
// The core idea in pure Go — a buffered channel IS a queue:
package main

func main() {
	jobs := make(chan string, 60) // buffer of 60 = the queue depth cap

	// start 3 workers draining the same channel
	for w := 1; w <= 3; w++ {
		go func(id int) {
			for to := range jobs { // blocks until a job arrives
				sendEmail(to)       // the slow work
			}
		}(w)
	}

	// PRODUCER: the API handler
	http.HandleFunc("POST /signup", func(w http.ResponseWriter, r *http.Request) {
		user := createUser(r)
		select {
		case jobs <- user.Email:            // enqueue
			w.WriteHeader(http.StatusAccepted) // 202
		default:
			http.Error(w, "busy", http.StatusServiceUnavailable) // 503 — full
		}
	})
	http.ListenAndServe(":3000", nil)
}

Going further

Real queues (Redis/BullMQ, RabbitMQ, AWS SQS, Kafka) survive a crash — an in-memory channel does not. Make handlers idempotent: queues deliver "at least once", so the same job can arrive twice (see the system-design capstone). Watch queue depth as your primary alarm — rising depth means add workers or you're about to fall behind. Send permanently-failing jobs to a dead-letter queue instead of retrying forever.
MODULE 07

Auth & JWT

HTTP forgets you between requests. Sessions fix that with server-side memory; JWTs fix it with math. At login, the server signs a token: header.payload.signature. The client sends it back on every request, and the server verifies the signature — no database lookup, no session store.

The payload is only encoded, not encrypted — anyone can read it. The signature is what makes it tamper-proof: change one byte and verification fails.

Try: walk all four steps, then hit "TAMPER" to edit the payload and watch the server reject the forged token.

🔧 Go deeper — sign, verify & protect routes with JWT

The stepper walked the concept; this is the real thing. Login signs a token; middleware verifies it on every protected route and rejects tampering — exactly the failure you triggered with the TAMPER button.

// npm i jsonwebtoken bcrypt
import jwt from "jsonwebtoken";
import bcrypt from "bcrypt";
const SECRET = process.env.JWT_SECRET; // long, random, never in git

app.post("/login", async (req, res) => {
  const user = await db.findByEmail(req.body.email);
  // compare against a SLOW salted hash, never a plaintext password
  if (!user || !(await bcrypt.compare(req.body.password, user.hash)))
    return res.status(401).json({ error: "bad credentials" });

  const token = jwt.sign(
    { sub: user.id, role: user.role },   // payload (readable by anyone!)
    SECRET,
    { expiresIn: "1h" }                  // exp claim
  );
  res.json({ token });
});

// middleware: verify the signature on every protected request
function auth(req, res, next) {
  const token = (req.headers.authorization || "").replace("Bearer ", "");
  try {
    req.user = jwt.verify(token, SECRET); // throws if tampered/expired
    next();
  } catch {
    res.status(401).json({ error: "invalid or expired token" });
  }
}

app.get("/profile", auth, (req, res) => res.json({ id: req.user.sub }));
// go get github.com/golang-jwt/jwt/v5 golang.org/x/crypto/bcrypt
var secret = []byte(os.Getenv("JWT_SECRET"))

func login(w http.ResponseWriter, r *http.Request) {
	u, _ := db.FindByEmail(readEmail(r))
	if u == nil || bcrypt.CompareHashAndPassword(u.Hash, readPass(r)) != nil {
		http.Error(w, "bad credentials", http.StatusUnauthorized) // 401
		return
	}
	tok := jwt.NewWithClaims(jwt.SigningMethodHS256, jwt.MapClaims{
		"sub":  u.ID,
		"role": u.Role,
		"exp":  time.Now().Add(time.Hour).Unix(),
	})
	signed, _ := tok.SignedString(secret)
	writeJSON(w, 200, map[string]string{"token": signed})
}

// middleware: reject anything without a valid signature
func auth(next http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		raw := strings.TrimPrefix(r.Header.Get("Authorization"), "Bearer ")
		_, err := jwt.Parse(raw, func(t *jwt.Token) (any, error) {
			return secret, nil // same key verifies the HMAC
		})
		if err != nil {
			http.Error(w, "invalid token", http.StatusUnauthorized)
			return
		}
		next(w, r)
	}
}

Going further

Hash passwords with bcrypt/argon2 — slow and salted, never MD5/SHA-1. Keep the secret out of git (env var or a secrets manager). Short-lived access tokens (~15 min) plus a longer refresh token limit the damage of a leak. JWTs can't be revoked before they expire, so for logout/ban you keep a small deny-list in Redis. For third-party login, this is what OAuth/OIDC hands you.

Scaling up vs scaling out

When traffic grows you have exactly two moves. Every architecture diagram you'll ever draw is some mix of them.

🔧 Go deeper — make an app horizontally scalable

Scaling out isn't a library — it's a rule: store no state in the process. Here's the difference between an app that can only scale up and one that scales out.

// ❌ CAN ONLY SCALE UP — state lives in this one process's memory.
// Add a 2nd server behind a load balancer and half of logins "forget" you.
const sessions = new Map();           // in-memory — dies with the process
app.post("/login", (req, res) => { sessions.set(sid, user); /* ... */ });

// ✅ SCALES OUT — state lives in a shared store every server can reach.
import Redis from "ioredis";
const redis = new Redis(process.env.REDIS_URL);
app.post("/login", async (req, res) => {
  await redis.set(`session:${sid}`, JSON.stringify(user), "EX", 3600);
});
// Now run N identical copies; the load balancer (FIG. 06) spreads traffic.
// Even simpler: put state in a JWT (FIG. 08) and store nothing server-side.

// Use every core on ONE box first (vertical), THEN add boxes (horizontal):
import cluster from "node:cluster";
import { cpus } from "node:os";
if (cluster.isPrimary) cpus().forEach(() => cluster.fork());
else app.listen(3000);
// Go serves every request on its own goroutine and uses all CPU cores
// out of the box (GOMAXPROCS) — so it scales UP for free.
//
// To scale OUT, the same rule applies: no per-process state.

// ❌ in-memory session map — breaks the moment you run a second instance
var sessions = map[string]User{}

// ✅ shared store — any instance can serve any request
func login(w http.ResponseWriter, r *http.Request) {
	u := authenticate(r)
	b, _ := json.Marshal(u)
	rdb.Set(r.Context(), "session:"+sid, b, time.Hour) // Redis
	// ...or issue a JWT and keep zero server-side state.
}

The mental model

Vertical (scale up): one bigger machine — simplest, but has a ceiling and a single point of failure. Horizontal (scale out): many stateless machines behind a load balancer — no ceiling, fault-tolerant, but needs shared state (Redis/DB) and orchestration. Real systems do both: max out one box, then add boxes. The whole system-design capstone in Security & Ops is this rule applied step by step.
